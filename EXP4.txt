import numpy as np

# Sales data for four quarters
sales_data = np.array([10000, 15000, 18000, 25000])
total_sales = np.sum(sales_data)

# Percentage increase from Q1 to Q4
percentage_increase = ((sales_data[3] - sales_data[0]) / sales_data[0]) * 100

print("Quarterly Sales:", sales_data)
print("Total Annual Sales:", total_sales)
print("Percentage Increase:", percentage_increase, "%")
