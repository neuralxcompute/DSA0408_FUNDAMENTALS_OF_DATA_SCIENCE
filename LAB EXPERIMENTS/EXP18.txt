# Experiment 18: Age and %Fat Analysis (Stats, Boxplot, Scatter Plot, Q-Q Plot)
import pandas as pd
import matplotlib.pyplot as plt
import scipy.stats as stats

data = {
    'age': [23, 23, 27, 27, 39, 41, 47, 49, 50,
            52, 54, 54, 56, 57, 58, 58, 60, 61],
    '%fat': [26.5, 9.5, 7.8, 17.8, 31.4, 25.9, 27.4, 27.2, 31.2,
             34.6, 42.5, 28.8, 33.4, 30.2, 34.1, 32.9, 41.2, 35.7]
}

df = pd.DataFrame(data)

# 1. Calculate Mean, Median, and Standard Deviation
print("Summary Statistics:")
for col in ['age', '%fat']:
    print(f"
--- {col} ---")
    print(f"Mean:   {df[col].mean():.2f}")
    print(f"Median: {df[col].median():.2f}")
    print(f"Std:    {df[col].std():.2f}")

# 2. Boxplots for age and %fat
fig, axes = plt.subplots(1, 2, figsize=(10, 4))
df.boxplot(column='age', ax=axes[0])
axes[0].set_title('Boxplot of Age')
df.boxplot(column='%fat', ax=axes[1])
axes[1].set_title('Boxplot of %Fat')
plt.tight_layout()
plt.show()

# 3. Scatter Plot of Age vs %Fat
plt.figure(figsize=(6, 4))
plt.scatter(df['age'], df['%fat'], color='purple', s=60)
plt.title('Scatter Plot: Age vs %Fat')
plt.xlabel('Age')
plt.ylabel('%Fat')
plt.grid(True)
plt.tight_layout()
plt.show()

# 4. Q-Q Plots for Age and %Fat
fig, axes = plt.subplots(1, 2, figsize=(10, 4))
stats.probplot(df['age'], dist="norm", plot=axes[0])
axes[0].set_title('Q-Q Plot of Age')
stats.probplot(df['%fat'], dist="norm", plot=axes[1])
axes[1].set_title('Q-Q Plot of %Fat')
plt.tight_layout()
plt.show()
