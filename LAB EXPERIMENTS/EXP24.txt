# Experiment 24: K-Nearest Neighbors (KNN) Classifier for Medical Condition Prediction
import numpy as np
from sklearn.neighbors import KNeighborsClassifier

# Sample training dataset: [fever_temp, cough_severity(1-10), fatigue_level(1-10)]
X_train = np.array([
    [98.6, 1, 2],
    [98.4, 2, 1],
    [99.0, 2, 3],
    [101.5, 7, 8],
    [102.0, 8, 9],
    [100.8, 6, 7],
    [98.7, 1, 1],
    [103.0, 9, 9]
])
# Labels: 0 = No Condition, 1 = Has Condition
y_train = np.array([0, 0, 0, 1, 1, 1, 0, 1])

# User input for new patient features and k
fever = float(input("Enter patient temperature (°F, e.g., 101.2): "))
cough = float(input("Enter cough severity (1-10): "))
fatigue = float(input("Enter fatigue level (1-10): "))
k = int(input("Enter value of k (number of neighbors, e.g., 3): "))

# Train KNN classifier and predict
knn = KNeighborsClassifier(n_neighbors=k)
knn.fit(X_train, y_train)

new_patient = np.array([[fever, cough, fatigue]])
prediction = knn.predict(new_patient)[0]

result = "Has Medical Condition (1)" if prediction == 1 else "No Medical Condition (0)"
print(f"
Predicted Outcome: {result}")
