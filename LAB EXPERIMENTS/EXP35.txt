# Experiment 35: Retail Store Customer Segmentation Using K-Means
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
from sklearn.preprocessing import StandardScaler

# Sample retail transaction data
transactions = pd.DataFrame({
    'customer_id': range(101, 113),
    'total_spent': [150, 200, 180, 850, 920, 780, 2100, 2400, 1950, 220, 890, 2250],
    'visit_frequency': [2, 3, 2, 10, 12, 9, 25, 28, 22, 4, 11, 26]
})

X = transactions[['total_spent', 'visit_frequency']]
X_scaled = StandardScaler().fit_transform(X)

# Apply K-Means clustering
kmeans = KMeans(n_clusters=3, random_state=42, n_init=10)
transactions['cluster'] = kmeans.fit_predict(X_scaled)

print("Segmented Customers:")
print(transactions)

# Plot clusters
plt.figure(figsize=(7, 5))
plt.scatter(transactions['visit_frequency'], transactions['total_spent'],
            c=transactions['cluster'], cmap='plasma', s=90, edgecolor='black')
plt.title('Retail Customer Segmentation (K-Means)')
plt.xlabel('Frequency of Visits')
plt.ylabel('Total Amount Spent ($)')
plt.colorbar(label='Cluster')
plt.grid(True)
plt.tight_layout()
plt.show()
