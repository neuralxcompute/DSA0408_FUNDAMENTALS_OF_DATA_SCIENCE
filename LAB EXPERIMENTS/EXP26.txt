# Experiment 26: Linear Regression for Housing Price Prediction
import numpy as np
from sklearn.linear_model import LinearRegression

# Sample dataset: [area_sqft, bedrooms, location_score(1-5)]
X = np.array([
    [1200, 2, 3],
    [1500, 3, 4],
    [1800, 3, 3],
    [2200, 4, 5],
    [2500, 4, 4],
    [3000, 5, 5]
])
y = np.array([210000, 290000, 310000, 430000, 460000, 580000])

# Train Linear Regression model
model = LinearRegression()
model.fit(X, y)

# User input for new house
area = float(input("Enter house area (sq ft): "))
bedrooms = int(input("Enter number of bedrooms: "))
location_score = float(input("Enter location rating (1-5): "))

predicted_price = model.predict([[area, bedrooms, location_score]])[0]
print(f"
Predicted House Price: ${predicted_price:,.2f}")
