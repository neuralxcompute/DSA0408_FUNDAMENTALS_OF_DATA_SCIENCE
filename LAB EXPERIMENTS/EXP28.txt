# Experiment 28: K-Means Clustering for Customer Segmentation
import numpy as np
from sklearn.cluster import KMeans

# Sample customer shopping data: [annual_spending($), visits_per_month]
X = np.array([
    [200, 2], [250, 3], [300, 2],
    [1200, 8], [1350, 9], [1100, 7],
    [3000, 15], [3200, 18], [2900, 14]
])

# Train K-Means model with 3 clusters
kmeans = KMeans(n_clusters=3, random_state=42, n_init=10)
kmeans.fit(X)

# User input for a new customer
spending = float(input("Enter customer annual spending ($): "))
visits = float(input("Enter customer visits per month: "))

cluster = kmeans.predict([[spending, visits]])[0]
print(f"
The new customer is assigned to Segment (Cluster): {cluster}")
