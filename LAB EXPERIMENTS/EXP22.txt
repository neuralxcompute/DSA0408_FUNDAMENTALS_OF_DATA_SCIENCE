# Experiment 22: Customer Reviews Average Rating & Confidence Interval (Pandas)
import os
import numpy as np
import pandas as pd
from scipy import stats

filename = "customer_reviews.csv"

# Create sample customer_reviews.csv if it does not exist
if not os.path.exists(filename):
    np.random.seed(42)
    ratings = np.random.choice([1, 2, 3, 4, 5], size=120, p=[0.05, 0.08, 0.17, 0.35, 0.35])
    pd.DataFrame({'rating': ratings}).to_csv(filename, index=False)

# Load customer reviews
df = pd.read_csv(filename)

mean_rating = df['rating'].mean()
std_rating = df['rating'].std()
n = len(df['rating'])
sem = std_rating / np.sqrt(n)

# 95% Confidence Interval
confidence = 0.95
ci_low, ci_high = stats.t.interval(confidence, df=n - 1, loc=mean_rating, scale=sem)

print(f"Total Reviews: {n}")
print(f"Average Rating: {mean_rating:.2f} / 5.00")
print(f"95% Confidence Interval for Population Mean Rating: ({ci_low:.2f}, {ci_high:.2f})")
